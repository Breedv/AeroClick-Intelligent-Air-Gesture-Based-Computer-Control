import eel
import os
import threading
from queue import Queue
import Gesture_Controller   # make sure this file exists in src folder


# ================== START GESTURE CONTROLLER ==================
def start_gesture_controller():
    """
    Starts the OpenCV + MediaPipe camera loop
    """
    Gesture_Controller.main()


# ================== CHATBOT CLASS ==================
class ChatBot:

    started = False
    userinputQueue = Queue()

    def isUserInput():
        return not ChatBot.userinputQueue.empty()

    def popUserInput():
        return ChatBot.userinputQueue.get()

    def close_callback(route, websockets):
        ChatBot.started = False
        os._exit(0)

    @eel.expose
    def getUserInput(msg):
        ChatBot.userinputQueue.put(msg)
        print("User:", msg)

    def close():
        ChatBot.started = False

    def addUserMsg(msg):
        eel.addUserMsg(msg)

    def addAppMsg(msg):
        eel.addAppMsg(msg)

    def start():
        print("APP STARTED")

        path = os.path.dirname(os.path.abspath(__file__))
        eel.init(path + r'\web', allowed_extensions=['.js', '.html'])

        # 🔥 START CAMERA IN BACKGROUND THREAD
        gesture_thread = threading.Thread(
            target=start_gesture_controller,
            daemon=True
        )
        gesture_thread.start()
        print("GESTURE CONTROLLER STARTED")

        try:
            eel.start(
                'index.html',
                mode='chrome',
                host='localhost',
                port=27005,
                block=False,
                size=(350, 480),
                position=(10, 100),
                disable_cache=True,
                close_callback=ChatBot.close_callback
            )

            ChatBot.started = True
            while ChatBot.started:
                eel.sleep(1.0)

        except Exception as e:
            print("ERROR:", e)


# ================== MAIN ==================
if __name__ == "__main__":
    ChatBot.start()
